cmake_minimum_required(VERSION 3.24)
project(jh_real_components LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(BLAZE_INSTALL OFF CACHE BOOL "" FORCE)
set(BLAZE_TESTS OFF CACHE BOOL "" FORCE)
set(BLAZE_BENCHMARK OFF CACHE BOOL "" FORCE)
add_subdirectory(${BLAZE_SOURCE} blaze)
set(Core_FOUND ON)
set(Blaze_FOUND ON)
set(JSONBINPACK_INSTALL OFF CACHE BOOL "" FORCE)
set(JSONBINPACK_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${BINPACK_SOURCE} binpack)
add_executable(jh_blaze_worker worker.cc)
target_link_libraries(jh_blaze_worker PRIVATE sourcemeta::blaze::compiler sourcemeta::blaze::evaluator sourcemeta::blaze::alterschema sourcemeta::core::json sourcemeta::core::jsonschema)
add_executable(jh_binpack binpack.cc)
target_link_libraries(jh_binpack PRIVATE sourcemeta::jsonbinpack::compiler sourcemeta::jsonbinpack::runtime sourcemeta::core::json sourcemeta::core::jsonschema)
